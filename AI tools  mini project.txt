<script>
const STORAGE_KEY = "studymate-planner-v1";

let subjects = [
  "Mathematics-I",
  "Engineering Physics",
  "Programming in C",
  "Engineering Chemistry"
];

let plan = [];
let completed = {};
let currentDay = 0;

const $ = id => document.getElementById(id);

function showMessage(message) {
  $("message").textContent = message;
}

function escapeHTML(value) {
  return String(value).replace(/[&<>"']/g, character => ({
    "&": "&amp;",
    "<": "&lt;",
    ">": "&gt;",
    '"': "&quot;",
    "'": "&#39;"
  })[character]);
}

function renderSubjects() {
  $("subjectList").innerHTML = subjects.map((subject, index) => `
    <span class="chip">
      ${escapeHTML(subject)}
      <button type="button"
        aria-label="Remove ${escapeHTML(subject)}"
        onclick="removeSubject(${index})">×</button>
    </span>
  `).join("");
}

function addSubject() {
  const value = $("subjectInput").value.trim();

  if (!value) {
    showMessage("Please enter a subject name.");
    return;
  }

  if (subjects.some(s => s.toLowerCase() === value.toLowerCase())) {
    showMessage("This subject has already been added.");
    return;
  }

  if (subjects.length >= 10) {
    showMessage("You can add up to 10 subjects.");
    return;
  }

  subjects.push(value);
  $("subjectInput").value = "";

  renderSubjects();
  saveData();
  showMessage("Subject added successfully.");
}

function removeSubject(index) {
  if (subjects.length <= 1) {
    showMessage("Keep at least one subject.");
    return;
  }

  subjects.splice(index, 1);
  renderSubjects();
  saveData();
  showMessage("Subject removed.");
}

function createTask(id, subject, title, detail, minutes, isBreak = false) {
  return {
    id,
    subject,
    title,
    detail,
    minutes,
    isBreak
  };
}

function generatePlan() {
  const dayCount = Number($("days").value);
  const hoursPerDay = Number($("hours").value);
  const totalMinutes = hoursPerDay * 60;
  const goal = $("goal").value;

  plan = [];
  completed = {};

  for (let day = 0; day < dayCount; day++) {
    const breakMinutes = 10;
    const studyMinutes = totalMinutes - breakMinutes;

    const taskCount = Math.min(
      subjects.length,
      hoursPerDay === 1 ? 1 : hoursPerDay === 2 ? 2 : 3
    );

    const baseMinutes = Math.floor(
      studyMinutes / (taskCount + 1)
    );

    const tasks = [];

    for (let i = 0; i < taskCount; i++) {
      const subject = subjects[(day + i) % subjects.length];

      let title;
      let detail;

      if (goal === "exam") {
        title = "Exam Preparation";
        detail = `Practise exam questions and review important topics in ${subject}.`;
      } else if (goal === "practice") {
        title = "Problem-Solving Practice";
        detail = `Solve 3–5 problems or write a small program related to ${subject}.`;
      } else if (goal === "revision") {
        title = "Revision Session";
        detail = `Review important concepts in ${subject} without looking at your notes.`;
      } else {
        const activities = [
          ["Learn Concepts", `Read and understand important concepts in ${subject}.`],
          ["Practice Questions", `Solve questions and examples from ${subject}.`],
          ["Quick Revision", `Revise key points and make short notes for ${subject}.`]
        ];

        [title, detail] = activities[day % activities.length];
      }

      tasks.push(createTask(
        `day${day}task${i}`,
        subject,
        title,
        detail,
        baseMinutes
      ));
    }

    const recapSubject = subjects[
      (day + taskCount) % subjects.length
    ];

    tasks.push(createTask(
      `day${day}recap`,
      recapSubject,
      "Daily Recap",
      `Summarize what you learned in ${recapSubject} and list any doubts.`,
      studyMinutes - baseMinutes * taskCount
    ));

    tasks.push(createTask(
      `day${day}break`,
      "",
      "Short Break",
      "Stretch, drink water, and rest your eyes.",
      breakMinutes,
      true
    ));

    plan.push({ tasks });
  }

  currentDay = 0;

  $("daySelect").innerHTML = plan.map((item, index) =>
    `<option value="${index}">Day ${index + 1}</option>`
  ).join("");

  $("daySelect").disabled = false;

  const name = $("studentName").value.trim();

  $("scheduleTitle").textContent = name
    ? `${name}'s Study Schedule`
    : "Your Study Schedule";

  $("scheduleSubtitle").textContent =
    `${dayCount}-day plan · ${hoursPerDay} hour(s) per day · ${subjects.length} subject(s)`;

  renderSchedule();
  updateStatistics();
  saveData();

  showMessage("Your study plan has been generated!");
}

function renderSchedule() {
  const day = plan[currentDay];

  if (!day) {
    $("scheduleContent").innerHTML =
      '<p class="subtitle">Generate a plan to view your tasks.</p>';
    return;
  }

  const studyTasks = day.tasks.filter(task => !task.isBreak);

  const finished = studyTasks.filter(
    task => completed[task.id]
  ).length;

  const percentage = Math.round(
    finished / Math.max(1, studyTasks.length) * 100
  );

  let html = `
    <div class="progress">
      <div style="display:flex;justify-content:space-between;gap:8px">
        <strong>Day ${currentDay + 1} Progress</strong>
        <span>${finished}/${studyTasks.length} tasks · ${percentage}%</span>
      </div>
      <div class="progress-bar">
        <div class="progress-fill" style="width:${percentage}%"></div>
      </div>
    </div>
  `;

  for (const task of day.tasks) {
    const isCompleted = Boolean(completed[task.id]);

    html += `
      <div class="task ${isCompleted ? "completed" : ""}">
        ${
          task.isBreak
            ? "<span>☕</span>"
            : `<input type="checkbox"
                aria-label="Mark ${escapeHTML(task.title)} complete"
                ${isCompleted ? "checked" : ""}
                onchange="toggleTask('${task.id}', this.checked)">`
        }

        <div>
          <strong>
            ${escapeHTML(task.title)}
            ${task.subject ? " · " + escapeHTML(task.subject) : ""}
          </strong>
          <div class="task-detail">
            ${escapeHTML(task.detail)}
          </div>
        </div>

        <span class="minutes">${task.minutes} min</span>
      </div>
    `;
  }

  html += `
    <p style="margin-top:18px">
      <button type="button" onclick="completeDay()">
        Complete Study Tasks
      </button>
      <button type="button"
        style="background:#edf0f8;color:#222;margin-top:6px"
        onclick="resetDay()">
        Reset Day
      </button>
    </p>
  `;

  $("scheduleContent").innerHTML = html;
}

function toggleTask(id, value) {
  completed[id] = value;
  renderSchedule();
  updateStatistics();
  saveData();
}

function completeDay() {
  if (!plan[currentDay]) return;

  plan[currentDay].tasks
    .filter(task => !task.isBreak)
    .forEach(task => completed[task.id] = true);

  renderSchedule();
  updateStatistics();
  saveData();
  showMessage("All study tasks marked complete!");
}

function resetDay() {
  if (!plan[currentDay]) return;

  plan[currentDay].tasks.forEach(task => {
    delete completed[task.id];
  });

  renderSchedule();
  updateStatistics();
  saveData();
  showMessage("Daily progress has been reset.");
}

function updateStatistics() {
  const allTasks = plan.flatMap(day => day.tasks)
    .filter(task => !task.isBreak);

  const finished = allTasks.filter(
    task => completed[task.id]
  ).length;

  const minutes = allTasks.reduce(
    (total, task) => total + task.minutes,
    0
  );

  $("doneCount").textContent = finished;
  $("hoursCount").textContent =
    (minutes / 60).toFixed(1) + "h";
}

function saveData() {
  try {
    localStorage.setItem(STORAGE_KEY, JSON.stringify({
      subjects,
      plan,
      completed,
      currentDay,
      name: $("studentName").value,
      hours: $("hours").value,
      days: $("days").value,
      goal: $("goal").value
    }));
  } catch (error) {
    showMessage("Browser storage is unavailable.");
  }
}

function loadData() {
  try {
    const saved = JSON.parse(
      localStorage.getItem(STORAGE_KEY)
    );

    if (saved) {
      if (Array.isArray(saved.subjects) && saved.subjects.length) {
        subjects = saved.subjects;
      }

      plan = Array.isArray(saved.plan) ? saved.plan : [];
      completed = saved.completed || {};
      currentDay = Math.min(
        saved.currentDay || 0,
        Math.max(0, plan.length - 1)
      );

      $("studentName").value = saved.name || "";
      $("hours").value = saved.hours || "3";
      $("days").value = saved.days || "7";
      $("goal").value = saved.goal || "balanced";
    }
  } catch (error) {
    console.log("No saved plan found.");
  }

  renderSubjects();

  if (plan.length) {
    $("daySelect").innerHTML = plan.map((day, index) =>
      `<option value="${index}">Day ${index + 1}</option>`
    ).join("");

    $("daySelect").disabled = false;
    $("daySelect").value = currentDay;

    const name = $("studentName").value.trim();

    $("scheduleTitle").textContent = name
      ? `${name}'s Study Schedule`
      : "Your Study Schedule";

    $("scheduleSubtitle").textContent =
      `${plan.length}-day plan · ${$("hours").value} hour(s) per day`;

    renderSchedule();
    updateStatistics();
  }
}

$("addButton").addEventListener("click", addSubject);

$("subjectInput").addEventListener("keydown", event => {
  if (event.key === "Enter") {
    event.preventDefault();
    addSubject();
  }
});

$("generateButton").addEventListener("click", generatePlan);

$("daySelect").addEventListener("change", event => {
  currentDay = Number(event.target.value);
  renderSchedule();
  saveData();
});

$("studentName").addEventListener("change", () => {
  if (plan.length) {
    const name = $("studentName").value.trim();

    $("scheduleTitle").textContent = name
      ? `${name}'s Study Schedule`
      : "Your Study Schedule";
  }

  saveData();
});

loadData();
</script>
